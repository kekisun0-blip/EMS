function formatStepLabel(i){
  const h = Math.floor(i / 4), m = (i % 4) * 15;
  return String(h).padStart(2, '0') + ':' + String(m).padStart(2, '0');
}
function emsChartXCategories(){
  const a = [];
  for(let i = 0; i < STEPS; i++){
    a.push(i % 4 === 0 ? formatAxisHour(Math.floor(i / 4)) : '');
  }
  return a;
}
function emsChartBaseTheme(){
  return {
    backgroundColor: 'transparent',
    textStyle: { color: '#94a3b8', fontFamily: 'Inter,IBM Plex Sans,system-ui,sans-serif' },
    animationDurationUpdate: 380,
    animationEasingUpdate: 'cubicOut'
  };
}
function disposeEmsMainChart(){
  if(emsEchartsMain){
    try{ emsEchartsMain.dispose(); }catch(e){}
    emsEchartsMain = null;
  }
  emsChartDrawStamp = { reveal: -1, tab: '', h: 0, sim: null };
}

function drawChart(){
  const host = document.getElementById('chart-echarts-host');
  if(!host){
    return;
  }
  if(typeof echarts === 'undefined'){
    host.innerHTML = '<p style="color:#64748b;padding:24px;text-align:center;font-size:12px;">' + emsT('图表库加载失败，请检查网络。', 'Chart library failed to load.') + '</p>';
    return;
  }
  const wrap = host.parentElement;
  const h = Math.max(220, (wrap && wrap.clientHeight) ? (wrap.clientHeight - 42) : 260);
  host.style.width = '100%';
  host.style.height = h + 'px';
  if(!simResult){
    disposeEmsMainChart();
    emsEchartsMain = echarts.init(host, null, { renderer: 'canvas' });
    emsEchartsMain.setOption({
      ...emsChartBaseTheme(),
      title: {
        text: emsT('运行仿真后显示曲线', 'Run simulation for charts'),
        left: 'center',
        top: 'center',
        textStyle: { color: '#64748b', fontSize: 14, fontWeight: 500 }
      }
    });
    return;
  }
  if(chartTab === 'predsoc' || chartTab === 'actsoc') chartTab = 'soc';
  const reveal = getChartRevealSteps();
  const simRef = simResult;
  if(!emsEchartsMain){
    emsEchartsMain = echarts.init(host, null, { renderer: 'canvas' });
  }
  if(emsEchartsMain && simRef &&
      reveal === emsChartDrawStamp.reveal &&
      chartTab === emsChartDrawStamp.tab &&
      simRef === emsChartDrawStamp.sim){
    if(Math.abs(h - emsChartDrawStamp.h) >= 2){
      emsEchartsMain.resize();
      emsChartDrawStamp.h = h;
    }
    return;
  }
  if(emsEchartsMain && Math.abs(h - emsChartDrawStamp.h) >= 2){
    emsEchartsMain.resize();
  }
  emsEchartsMain.setOption(buildMainEchartsOption(), { notMerge: true, lazyUpdate: false });
  emsChartDrawStamp = { reveal: reveal, tab: chartTab, h: h, sim: simRef };
}
